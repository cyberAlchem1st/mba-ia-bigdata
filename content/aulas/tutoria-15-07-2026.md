## Exemplos de código

```python
pontos3d = [ [1.0, 1.0, 1.0],
             [0.0, 1.0, 3.0],
             [2.0, 2.0, 2.0],
             [0.0, 0.0, 0.0],]

```

```python
resultados = []
linha = []

for i in range(len(pontos3d)):
  for j in range(len(pontos3d)):
      distancia = 0
      p1 = pontos3d[i]
      p2 = pontos3d[j]

      if i == j:
        distancia = 0
      else:
        distancia = abs(p1[0] - p2[0]) + abs(p1[1] - p2[1]) + abs(p1[2] - p2[2])

      linha.append(distancia)
  resultados.append(linha)
  linha = []

```

```python
resultados2 = []
linha = []
for ponto in pontos3d:
  for ponto2 in pontos3d:
    if ponto == ponto2:
      linha.append(0)
    else:
      distancia = abs(ponto[0] - ponto2[0]) + abs(ponto[1] - ponto2[1]) + abs(ponto[2] - ponto2[2])
      linha.append(distancia)
  resultados2.append(linha)
  linha = []

```
