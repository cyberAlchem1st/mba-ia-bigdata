A = ['a', 'b']
B = [10, 20, 30]

prod_cart = {(a,b) for a in A for b in B}
print("Com comprehension:", prod_cart)
print(type(prod_cart))
# com for
prod_cart_for = set()
for a in A:
    for b in B:
        prod_cart_for.add((a,b))

print("Com for:          ", prod_cart_for)
